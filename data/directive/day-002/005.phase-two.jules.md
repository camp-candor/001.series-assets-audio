  description: string;

  schema_version: '1.0.0';
  series_scope: string;
  cues: Record<string, FoleyCueDefinition>;
}

export interface FoleyResolutionResult {
  valid: boolean;
  cueSlug: string;
  selectedVariantIndex: number;
  variant: FoleyVariantEntry | null;

#### 2. Foley Domain Buzzers (`02.foley.unit/buz/foley.buzz.ts` & `02.foley.unit/buz/poolBalancer.buzz.ts`)
